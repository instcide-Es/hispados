# Solo esa URL, solo POST, como mucho 6 MB, con su propio pool de PHP; el
# guion está fuera de la carpeta de la web (no se puede descargar su fuente).
location = /termo/subir.php {
    limit_except POST { deny all; }
    client_max_body_size 6m;
    client_body_timeout 60s;
    fastcgi_pass unix:/tmp/php-cgi-83-hispados-termo.sock;
    include fastcgi_params;
    fastcgi_param SCRIPT_FILENAME /www/hispados-termo/subir.php;
    fastcgi_param SCRIPT_NAME /termo/subir.php;
    access_log /www/wwwlogs/hispados-termo.log;
}
location ^~ /termo/ { return 404; }
